# Binary-only Morse Micro module for nRF54L15 MM6108/MM8108 profiles.
if(CONFIG_DT_HAS_MORSE_MM8108_ENABLED)
  set(MORSE_BINARY_CHIP mm8108)
elseif(CONFIG_DT_HAS_MORSE_MM6108_ENABLED)
  set(MORSE_BINARY_CHIP mm6108)
else()
  message(FATAL_ERROR "No supported Morse MM6108/MM8108 devicetree node is enabled")
endif()
set(MORSE_BINARY_PROFILE "${MORSE_BINARY_CHIP}")
if(MORSE_BINARY_CHIP STREQUAL "mm6108" AND MORSE_BINARY_VARIANT STREQUAL "sense")
  set(MORSE_BINARY_PROFILE "mm6108-sense")
endif()
set(MORSE_BINARY_ARCHIVE "${CMAKE_CURRENT_LIST_DIR}/zephyr/blobs/lib/${MORSE_BINARY_PROFILE}/arm-cortex-m33f/libmorse.a")
if(CONFIG_WIFI_MORSE_SM)
  if(NOT EXISTS "${MORSE_BINARY_ARCHIVE}")
    message(FATAL_ERROR "Missing packaged Morse archive: ${MORSE_BINARY_ARCHIVE}")
  endif()
  add_library(libmorse STATIC IMPORTED GLOBAL)
  set_target_properties(libmorse PROPERTIES
    IMPORTED_LOCATION "${MORSE_BINARY_ARCHIVE}"
    INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_LIST_DIR}/include")
  target_compile_definitions(libmorse INTERFACE MMOSAL_NOASSERT MMOSAL_NO_DEBUGLOG)
  zephyr_include_directories("${CMAKE_CURRENT_LIST_DIR}/include")
  zephyr_compile_definitions(MMOSAL_NOASSERT MMOSAL_NO_DEBUGLOG)
endif()
